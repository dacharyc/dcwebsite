{{- /* Shared social-card values. description: frontmatter description, else the page's own summary (microblog posts have no description), else the site's. image: frontmatter image, else the first attached microblog image, else the author avatar for microblog posts, else the site hero. card: a square "summary" card when the image is the avatar, otherwise a large image card. */ -}} {{- $desc := .Description -}} {{- if and (not $desc) .IsPage }}{{ $desc = .Summary | plainify | htmlUnescape | truncate 200 }}{{ end -}} {{- if not $desc }}{{ $desc = .Site.Params.description }}{{ end -}} {{- $img := .Params.image -}} {{- if and (not $img) .Params.images }}{{ $img = (index .Params.images 0).src }}{{ end -}} {{- $card := "summary_large_image" -}} {{- if .IsHome }}{{ $img = .Site.Params.hero.hero__image }}{{ end -}} {{- if and (not $img) (eq .Type "microblog") }}{{ $img = .Site.Params.author.author__avatar }}{{ $card = "summary" }}{{ end -}} {{- if not $img }}{{ $img = .Site.Params.hero.hero__image }}{{ end -}} {{ with .OutputFormats.Get "rss" -}} {{ printf `` .Rel .MediaType.Type .Permalink $.Site.Title | safeHTML }} {{ end }} {{ with .Site.Params.author.author__mastodon }}{{ end }} {{ with resources.Get "sass/main.scss" | toCSS | minify | fingerprint }} {{ end }}
